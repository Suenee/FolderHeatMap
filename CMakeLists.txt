cmake_minimum_required(VERSION 3.24)
project(FolderHeatMap VERSION 1.51 LANGUAGES C CXX)

if(NOT WIN32)
    message(FATAL_ERROR "FolderHeatMap is a Windows Total Commander plugin.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "FolderHeatMap targets Total Commander x64. Configure an x64 build.")
endif()
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/vendor/sqlite/sqlite3.c")
    message(FATAL_ERROR "SQLite amalgamation is missing. Run upgrade.cmd first.")
endif()

set(FHM_ENGINE_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated_engine")
set(FHM_ENGINE_GENERATED_CPP "${FHM_ENGINE_GENERATED_DIR}/EngineApp.1.11.cpp")
file(MAKE_DIRECTORY "${FHM_ENGINE_GENERATED_DIR}")
execute_process(COMMAND "${CMAKE_COMMAND}" "-DINPUT=${CMAKE_SOURCE_DIR}/src/EngineApp.cpp" "-DOUTPUT=${FHM_ENGINE_GENERATED_CPP}" -P "${CMAKE_SOURCE_DIR}/cmake/GenerateOptimizedEngine.cmake" RESULT_VARIABLE FHM_ENGINE_GENERATE_RESULT)
if(NOT FHM_ENGINE_GENERATE_RESULT EQUAL 0 OR NOT EXISTS "${FHM_ENGINE_GENERATED_CPP}")
    message(FATAL_ERROR "Could not generate the guarded FolderHeatMap engine source.")
endif()
execute_process(COMMAND "${CMAKE_COMMAND}" "-DINPUT=${FHM_ENGINE_GENERATED_CPP}" -P "${CMAKE_SOURCE_DIR}/cmake/InjectDeletionDiagnostics.cmake" RESULT_VARIABLE FHM_DIAG_INJECT_RESULT)
if(NOT FHM_DIAG_INJECT_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not inject FolderHeatMap safe deletion lifecycle.")
endif()
execute_process(COMMAND "${CMAKE_COMMAND}" "-DINPUT=${FHM_ENGINE_GENERATED_CPP}" -P "${CMAKE_SOURCE_DIR}/cmake/InjectCanonicalIdentityDiagnostics.cmake" RESULT_VARIABLE FHM_IDENTITY_DIAG_RESULT)
if(NOT FHM_IDENTITY_DIAG_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not inject FolderHeatMap canonical identity diagnostics.")
endif()
execute_process(COMMAND "${CMAKE_COMMAND}" "-DINPUT=${FHM_ENGINE_GENERATED_CPP}" -P "${CMAKE_SOURCE_DIR}/cmake/CoalesceDeleteQueue.cmake" RESULT_VARIABLE FHM_DELETE_COALESCE_RESULT)
if(NOT FHM_DELETE_COALESCE_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not inject FolderHeatMap delete queue coalescing.")
endif()
execute_process(COMMAND "${CMAKE_COMMAND}" "-DINPUT=${FHM_ENGINE_GENERATED_CPP}" -P "${CMAKE_SOURCE_DIR}/cmake/ProtectSameVolumeMoves.cmake" RESULT_VARIABLE FHM_MOVE_GUARD_RESULT)
if(NOT FHM_MOVE_GUARD_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not inject FolderHeatMap same-volume move protection.")
endif()
execute_process(COMMAND "${CMAKE_COMMAND}" "-DINPUT=${FHM_ENGINE_GENERATED_CPP}" -P "${CMAKE_SOURCE_DIR}/cmake/ProtectRapidMoveRoundTrips.cmake" RESULT_VARIABLE FHM_ROUNDTRIP_GUARD_RESULT)
if(NOT FHM_ROUNDTRIP_GUARD_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not inject FolderHeatMap rapid MOVE arrival-side reconciliation.")
endif()
execute_process(COMMAND "${CMAKE_COMMAND}" "-DINPUT=${FHM_ENGINE_GENERATED_CPP}" -P "${CMAKE_SOURCE_DIR}/cmake/InjectFileWriteTracking.cmake" RESULT_VARIABLE FHM_FILE_WRITE_RESULT)
if(NOT FHM_FILE_WRITE_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not inject FolderHeatMap file write tracking.")
endif()
execute_process(COMMAND "${CMAKE_COMMAND}" "-DINPUT=${FHM_ENGINE_GENERATED_CPP}" -P "${CMAKE_SOURCE_DIR}/cmake/InjectTcNavigationMonitor.cmake" RESULT_VARIABLE FHM_TC_NAV_RESULT)
if(NOT FHM_TC_NAV_RESULT EQUAL 0)
    message(FATAL_ERROR "Could not inject FolderHeatMap independent Total Commander navigation monitor.")
endif()

add_library(FolderHeatMap SHARED src/FolderHeatMapCounterOnly.cpp)
target_include_directories(FolderHeatMap PRIVATE src)
target_compile_features(FolderHeatMap PRIVATE cxx_std_20)
target_compile_definitions(FolderHeatMap PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(FolderHeatMap PROPERTIES PREFIX "" SUFFIX ".wdx64")

add_executable(FolderHeatMapEngine "${FHM_ENGINE_GENERATED_CPP}" src/DiagnosticWatcher.cpp src/FolderIdentity.cpp src/Database.cpp src/DatabaseBaseline.cpp src/ResetDatabase.cpp src/RuntimeCacheDatabase.cpp src/LifecycleDatabase.cpp src/TrackedLookupDatabase.cpp src/LifecycleBatchDatabase.cpp src/Lifecycle.cpp src/Settings.cpp vendor/sqlite/sqlite3.c)
target_include_directories(FolderHeatMapEngine PRIVATE src vendor/sqlite)
target_compile_features(FolderHeatMapEngine PRIVATE cxx_std_20)
target_compile_definitions(FolderHeatMapEngine PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
add_dependencies(FolderHeatMap FolderHeatMapEngine)

set(FHM_CONFIG_ICON_B64 "${CMAKE_SOURCE_DIR}/assets/FolderHeatMapConfig.ico.b64")
set(FHM_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated")
set(FHM_CONFIG_ICON "${FHM_GENERATED_DIR}/FolderHeatMapConfig.ico")
set(FHM_CONFIG_RC "${FHM_GENERATED_DIR}/FolderHeatMapConfig.rc")
file(MAKE_DIRECTORY "${FHM_GENERATED_DIR}")
execute_process(COMMAND powershell.exe -NoProfile -ExecutionPolicy Bypass -Command "[IO.File]::WriteAllBytes('${FHM_CONFIG_ICON}', [Convert]::FromBase64String([IO.File]::ReadAllText('${FHM_CONFIG_ICON_B64}').Trim()))" RESULT_VARIABLE FHM_ICON_DECODE_RESULT)
if(NOT FHM_ICON_DECODE_RESULT EQUAL 0 OR NOT EXISTS "${FHM_CONFIG_ICON}")
    message(FATAL_ERROR "Could not decode FolderHeatMap configurator icon.")
endif()
file(WRITE "${FHM_CONFIG_RC}" "IDI_FHM_CONFIG ICON \"${FHM_CONFIG_ICON}\"\n")

add_executable(FolderHeatMapConfig WIN32 src/ConfigAppCompile.cpp src/LoggingConfigExtension.cpp src/Settings.cpp "${FHM_CONFIG_RC}")
target_include_directories(FolderHeatMapConfig PRIVATE src)
target_compile_features(FolderHeatMapConfig PRIVATE cxx_std_20)
target_compile_definitions(FolderHeatMapConfig PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(FolderHeatMapConfig PRIVATE comdlg32 shell32 advapi32 comctl32)

add_executable(FolderHeatMapReset WIN32 src/ResetApp.cpp src/ResetDatabase.cpp src/Database.cpp src/FolderIdentity.cpp vendor/sqlite/sqlite3.c)
target_include_directories(FolderHeatMapReset PRIVATE src vendor/sqlite)
target_compile_features(FolderHeatMapReset PRIVATE cxx_std_20)
target_compile_definitions(FolderHeatMapReset PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
target_link_libraries(FolderHeatMapReset PRIVATE shell32 advapi32)
